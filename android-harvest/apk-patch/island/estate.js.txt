function lagoon(g, t) {
  const [x, y, rx, ry] = LAGOON;
  g.fillStyle = '#FFF4D2'; g.beginPath(); g.ellipse(x, y, rx + 8, ry + 7, -0.08, 0, TAU); g.fill();
  const gr = g.createLinearGradient(0, y - ry, 0, y + ry); gr.addColorStop(0, '#8EEBE6'); gr.addColorStop(0.5, '#3FC9E0'); gr.addColorStop(1, '#2A9FD6');

  g.strokeStyle = 'rgba(255,255,255,0.75)'; g.lineWidth = 2; g.beginPath(); g.ellipse(x, y, rx - 3 + Math.sin(t * 1.4) * 1.5, ry - 3 + Math.sin(t * 1.4) * 1, -0.08, 0, TAU); g.stroke();

  for (let i = 0; i < 6; i++) { const ox = ((i * 37 + t * 6) % (rx * 1.4)) - rx * 0.7, oy = -ry * 0.5 + (i % 3) * ry * 0.4; g.beginPath(); g.moveTo(x + ox - 8, y + oy); g.quadraticCurveTo(x + ox, y + oy - 2, x + ox + 8, y + oy); g.stroke(); }
}
